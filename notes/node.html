1. Non-Blocking I/O & Libuv Event Loop

Node.js runs JavaScript on the V8 engine and uses Libuv thread pool for asynchronous file, network, and database operations. This single-threaded event loop design handles tens of thousands of concurrent client connections efficiently.

server.js 
const express = require('express');
const app = express();

app.use(express.json());

// In-Memory Skill Swap Store
let skills = [
    { id: 1, title: 'Fullstack Dev', category: 'Tech', points: 30 }
];

app.get('/api/skills', (req, res) => {
    res.json({ success: true, count: skills.length, data: skills });
});

app.post('/api/post-skill', (req, res) => {
    const newSkill = { id: Date.now(), ...req.body };
    skills.push(newSkill);
    res.status(201).json({ message: "Skill Shared! Earned +30 Points", skill: newSkill });
});

app.listen(3000, () => console.log('Node Server Running on port 3000'));

2. Express Middleware Pipeline & Error Handling

Express applications execute middleware functions sequentially in a request pipeline. Custom middleware handles authentication, logging, CORS, and centralized error catching.

authMiddleware.js 
function authMiddleware(req, res, next) {
    const token = req.headers.authorization;
    if (!token) return res.status(401).json({ error: "Access Denied. Token Missing." });
    
    try {
        req.user = { username: "Naveen", role: "Swapper" };
        next(); // Pass control to next handler
    } catch (err) {
        res.status(403).json({ error: "Invalid Authentication Token" });
    }
}
